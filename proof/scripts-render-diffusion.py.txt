import asyncio,json
from pathlib import Path
from playwright.async_api import async_playwright
ROOT=Path(__file__).resolve().parents[1]
async def main():
 async with async_playwright() as p:
  b=await p.chromium.launch(executable_path='/usr/bin/chromium',args=['--no-sandbox','--disable-dev-shm-usage'])
  page=await b.new_page(accept_downloads=True);logs=[]
  page.on('console',lambda m: logs.append(m.text));page.on('pageerror',lambda e: logs.append(str(e)))
  await page.goto('http://127.0.0.1:5173/diffusion/index.html');await page.wait_for_function('window.studio?.ready',timeout=90000)
  print(await page.evaluate('({duration:studio.composition.duration,isolated:crossOriginIsolated})'),flush=True)
  async with page.expect_download(timeout=600000) as di:await page.click('#render')
  download=await di.value;await download.save_as(ROOT/'output/tsmc-diffusion-native.mp4')
  result=await page.evaluate('window.renderResult');print(result,flush=True)
  (ROOT/'evidence/diffusion-render.json').write_text(json.dumps({'result':result,'logs':logs},indent=2));await b.close()
asyncio.run(main())
